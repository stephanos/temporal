package transcript

import (
	"encoding/hex"
	"encoding/json"
	"os"
	"path/filepath"
	"strings"
	"testing"

	nativebuild "go.temporal.io/server/tools/gomad/internal/backends/native/build"
	"go.temporal.io/server/tools/gomad/internal/backends/target"
)

func TestRenameNamespacePinAudit(t *testing.T) {
	output := os.Getenv("GOMAD_RENAME_PIN_AUDIT_OUTPUT")
	if output == "" || !filepath.IsAbs(output) {
		t.Fatal("GOMAD_RENAME_PIN_AUDIT_OUTPUT must name an absolute external file")
	}
	type pair struct {
		Before string `json:"before"`
		After  string `json:"after"`
	}
	type adapter struct {
		Before AdapterAnchors `json:"before"`
		After  AdapterAnchors `json:"after"`
	}
	report := struct {
		Templates map[string]pair              `json:"templates"`
		Profiles  map[string]map[string]string `json:"profiles"`
		Adapters  map[string]adapter           `json:"adapters"`
	}{
		Templates: map[string]pair{
			"gomadLibcAdapterSHA256":      {gomadLibcAdapterSHA256, digestBytes([]byte(gomadLibcAdapterSource))},
			"gomadLibcLinuxAdapterSHA256": {gomadLibcLinuxAdapterSHA256, digestBytes([]byte(gomadLibcLinuxAdapterSource))},
		},
		Profiles: map[string]map[string]string{},
		Adapters: map[string]adapter{},
	}
	defer func() {
		data, err := json.MarshalIndent(report, "", "  ")
		if err != nil {
			t.Fatal(err)
		}
		if err := os.WriteFile(output, append(data, '\n'), 0o600); err != nil {
			t.Fatal(err)
		}
	}()
	for _, platform := range []string{"darwin/arm64", "linux/amd64"} {
		goos, goarch, _ := strings.Cut(platform, "/")
		profile := mustSpec(profileDefinition{
			name:                  Deterministic,
			target:                TargetContract{GoVersion: generatedBoundaryGoVersion, GOOS: goos, GOARCH: goarch},
			implementationFamily:  "gomad.deterministic-io/v1",
			implementationVersion: deterministicImplementationVersion,
			adapters:              deterministicAdapters,
		})
		frame, err := encodeBootstrapFrame(profile, target.Prepared{SHA256: "sha256:aaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaaa", Argv: []string{"gomad-target", "argument"}}, "sha256:bbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbbb", 42)
		if err != nil {
			t.Fatal(err)
		}
		report.Profiles[platform] = map[string]string{
			"name": profile.Name(), "implementation_sha256": string(profile.ImplementationSHA256()),
			"inventory_sha256": string(profile.InventorySHA256()), "inventory": string(profile.Inventory()),
			"frame_hex":                    hex.EncodeToString(frame),
			"before_implementation_sha256": profileGoldens[platform].implementationSHA256,
			"before_inventory_sha256":      profileGoldens[platform].inventorySHA256,
			"before_frame_hex":             profileGoldens[platform].frameHex,
		}
	}
	goCommand, cache := portableAdapterGo(t)
	for _, definition := range deterministicAdapters.definitions {
		t.Run(definition.identity.Module, func(t *testing.T) {
			identity := definition.identity
			original := portableAdapterModule(t, cache, identity.Module, identity.Version)
			replacement := filepath.Join(t.TempDir(), "replacement")
			entry := adapter{After: AdapterAnchors{
				Version: identity.Version, Sum: identity.Sum,
				PreparedSourceSetSHA256: map[string]string{},
			}}
			var err error
			entry.After.OriginalSourceInventorySHA256, err = digestAdapterSourceInventory(original)
			if err != nil {
				t.Fatal(err)
			}
			var derived *rewrittenModule
			if spec := definition.implementation.rewritten; spec != nil {
				entry.Before = pinnedAnchors(*spec)
				entry.After.PreparedPackage = spec.preparedPackage
				replacements := map[string][]byte{}
				candidate := *spec
				candidate.rewrites = append([]sourceRewrite(nil), spec.rewrites...)
				for index, rewrite := range spec.rewrites {
					contents, err := readAdapterSource(spec.module, original, rewrite.path)
					if err != nil {
						t.Fatal(err)
					}
					if digestBytes(contents) != rewrite.sourceSHA256 {
						t.Fatalf("upstream source changed: %s", rewrite.path)
					}
					inputPath := rewrite.path
					if rewrite.base != "" {
						inputPath = rewrite.base
						contents, err = readAdapterSource(spec.module, original, rewrite.base)
						if err != nil {
							t.Fatal(err)
						}
						if digestBytes(contents) != rewrite.baseSHA256 {
							t.Fatalf("upstream base changed: %s", rewrite.base)
						}
					}
					result, err := applyAdapterAnchors(spec.module, inputPath, rewrite.rewrites, contents)
					if err != nil {
						t.Fatal(err)
					}
					replacements[rewrite.path] = result
					candidate.rewrites[index].replacementSHA256 = digestBytes(result)
					entry.After.Rewrites = append(entry.After.Rewrites, rewriteAnchors(candidate.rewrites[index]))
				}
				if err := copyAdapterModule(original, replacement, replacements, defaultAdapterCopyLimits); err != nil {
					t.Fatal(err)
				}
				derived = &candidate
			} else {
				if identity.Module != libcModulePath {
					t.Fatalf("unhandled custom adapter %s", identity.Module)
				}
				entry.Before.PreparedSourceSetSHA256 = libcPreparedSourceSetSHA256ByHost
				entry.After.PreparedPackage = libcModulePath
				replacements, _, err := rewriteLibcModule(original)
				if err != nil {
					t.Fatal(err)
				}
				if _, err := copyLibcModule(original, replacement, replacements); err != nil {
					t.Fatal(err)
				}
				for _, name := range sortedKeys(replacements) {
					entry.After.Rewrites = append(entry.After.Rewrites, RewriteAnchors{Path: name, ReplacementSHA256: digestBytes(replacements[name])})
				}
			}
			entry.After.ReplacementSourceInventorySHA256, err = digestAdapterSourceInventory(replacement)
			if err != nil {
				t.Fatal(err)
			}
			packageDirectory := replacement
			if entry.After.PreparedPackage != identity.Module {
				packageDirectory = filepath.Join(replacement, filepath.FromSlash(strings.TrimPrefix(entry.After.PreparedPackage, identity.Module+"/")))
			}
			for _, platform := range []string{"darwin/arm64", "linux/amd64"} {
				goos, goarch, _ := strings.Cut(platform, "/")
				entry.After.PreparedSourceSetSHA256[platform], err = nativebuild.AdapterPreparedSourceSetSHA256(t.Context(), goCommand, packageDirectory, entry.After.PreparedPackage, goos, goarch)
				if err != nil {
					t.Fatal(err)
				}
			}
			if derived != nil {
				derived.cacheElements = []string{original}
				derived.originalInventorySHA256 = entry.After.OriginalSourceInventorySHA256
				derived.replacementInventorySHA256 = entry.After.ReplacementSourceInventorySHA256
				derived.preparedSourceSetSHA256ByHost = entry.After.PreparedSourceSetSHA256
				if _, err := prepareRewrittenModule("", t.TempDir(), identity, *derived); err != nil {
					t.Fatal(err)
				}
			}
			report.Adapters[identity.Module] = entry
		})
	}
	if len(report.Adapters) != len(deterministicAdapters.definitions) {
		t.Errorf("computed %d of %d adapters", len(report.Adapters), len(deterministicAdapters.definitions))
	}
}
