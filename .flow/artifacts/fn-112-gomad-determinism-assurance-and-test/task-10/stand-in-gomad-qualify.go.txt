// Command gomadstub stands in for `gomad qualify --json` on a host the patched
// toolchain does not support. It writes a real qualification report whose
// repetitions carry real diagnostic traces; STUB_DIVERGE=1 perturbs the last
// repetition's seeded-stream draw counter.
package main

import (
	"encoding/binary"
	"fmt"
	"os"
	"path/filepath"
	"strconv"
	"strings"
	"time"

	"go.temporal.io/server/tools/gomad3/choice"
	"go.temporal.io/server/tools/gomad3/deterministicio"
	"go.temporal.io/server/tools/gomad3/qualification"
	"go.temporal.io/server/tools/gomad3/record"
	"go.temporal.io/server/tools/gomad3/runner"
)

func flag(args []string, name string) string {
	for _, a := range args {
		if v, ok := strings.CutPrefix(a, name+"="); ok {
			return v
		}
	}
	return ""
}

func main() {
	args := os.Args[1:]
	if len(args) == 0 || args[0] != "qualify" {
		fmt.Fprintln(os.Stderr, "stub supports qualify only")
		os.Exit(2)
	}
	seed, _ := strconv.ParseUint(flag(args, "--seed"), 10, 64)
	repeat, _ := strconv.ParseUint(flag(args, "--repeat"), 10, 64)
	artifacts := flag(args, "--artifacts")
	var runs []qualification.QualificationExecution
	for i := uint64(0); i < repeat; i++ {
		start := time.Now()
		draw := byte(0)
		if os.Getenv("STUB_DIVERGE") == "1" && i == repeat-1 {
			draw = 1
		}
		data := make([]byte, 160)
		copy(data, []byte{'G', 'O', 'M', 'A', 'D', 'D', 'G', 1})
		binary.BigEndian.PutUint32(data[8:12], 1)
		data[12] = 1
		binary.BigEndian.PutUint64(data[16:24], 160)
		binary.BigEndian.PutUint64(data[24:32], 160)
		binary.BigEndian.PutUint64(data[32:40], 1)
		data[143] = draw
		trace, err := choice.DecodeDiagnosticTrace(data)
		if err != nil {
			panic(err)
		}
		campaign := filepath.Join(artifacts, fmt.Sprintf("campaign-%d", i))
		path := filepath.Join(campaign, "diagnostics", "00000000000000000000.bin")
		_ = os.MkdirAll(filepath.Dir(path), 0o700)
		if err := os.WriteFile(path, data, 0o600); err != nil {
			panic(err)
		}
		time.Sleep(20 * time.Millisecond)
		ref := &runner.DiagnosticTraceReference{Path: path, DiagnosticEvidence: runner.DiagnosticEvidence{Profile: choice.DiagnosticProfile, SHA256: record.SHA256FromSum(trace.SHA256), Records: 1}}
		ev := runner.ExecutionEvidence{
			Schema: runner.ExecutionEvidenceSchema, Seed: record.Uint64String(seed), RunnerBuild: "sha256:stub-runner",
			Toolchain:   record.Toolchain{GoVersion: "go1.27.1-stub", BuildKey: "stub", TargetGOOS: "linux", TargetGOARCH: "arm64"},
			Target:      record.Target{Kind: "go-test", Source: "./basic/concurrency", SHA256: "sha256:stub-target", Size: 1, Argv: []string{"gomad3-target"}, BuildTags: []string{}, Adapters: []record.TargetAdapter{}, Compatibility: []record.CompatibilityPack{}},
			IOProfile:   deterministicio.Contract{Name: "deterministic", ImplementationSHA256: "sha256:io", InventorySHA256: "sha256:inventory"},
			Environment: []record.Environment{{Name: "GOMADSEED", Value: strconv.FormatUint(seed, 10)}},
			Outcome:     runner.OutcomeEvidence{Domain: "success", Reason: "success", Termination: "exit"}, GroupGone: true,
			Stdout: record.Stream{FullSHA256: "sha256:stdout"}, Stderr: record.Stream{FullSHA256: "sha256:stderr"},
			IOTranscriptSHA256: "sha256:transcript", IOTranscriptRecords: 1, IOTranscriptComplete: true,
			SemanticCoverage: deterministicio.SemanticCoverage{Schema: deterministicio.SemanticCoverageSchema, Digest: "sha256:coverage", Probes: []string{}},
			Diagnostics: &ref.DiagnosticEvidence,
		}
		runs = append(runs, qualification.QualificationExecution{CampaignPath: campaign, WallElapsedNanos: uint64(time.Since(start)), Evidence: ev, Diagnostics: ref})
		_ = qualification.WriteProgressEvent(os.Stdout, i+1, repeat)
	}
	report, err := qualification.BuildQualificationReport(qualification.QualificationInput{Command: append([]string{"gomad"}, args...), Executions: runs})
	if err != nil {
		panic(err)
	}
	path, err := qualification.WriteQualificationReport(artifacts, report)
	if err != nil {
		panic(err)
	}
	_ = qualification.WriteResultEvent(os.Stdout, report, path)
	os.Exit(qualification.ExitStatus(qualification.ClassifyQualification(report)))
}
