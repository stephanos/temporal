type ArtifactInput struct {
	Manifest       record.ExecutionRecord
	TargetPath     string
	Stdout         []byte
	Stderr         []byte

	ChoiceTrace    []byte
	ReadOnlyMounts *readonlymount.CapturedInputs
	World          record.WorldPayloads
	Simulation     *SimulationPayloads
}

type CapacityError struct {
	Required uint64
	Maximum  uint64
}

type Payload struct {
	Path       string
	Mode       os.FileMode
	SourcePath string
	Data       []byte

	Size       record.Uint64String
}

type Publication struct {
	Record   record.ExecutionRecord
	Payloads []Payload
}

type SimulationPayloads struct {
	Plan   []byte

}

	Key          StoreKey
}
