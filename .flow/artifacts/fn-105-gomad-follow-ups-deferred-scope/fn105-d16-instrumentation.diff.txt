+package contextutil
+
+import (
+	"context"
+	"fmt"
+	"os"
+	"time"
+)
+
+var d16Epoch = time.Date(2000, 1, 1, 0, 0, 0, 0, time.UTC)
+
+// D16Probe prints, for a context whose deadline is at most 4.5s away, the ticked
+// time.Now reading, the forward offset (time.Now minus the timer clock), the
+// timer-clock reading, and the deadline. A deadline carrying a monotonic reading
+// fires when the timer clock reaches the deadline's own numeric value.
+func D16Probe(site string, label string, ctx context.Context, err error) {
+	dl, ok := ctx.Deadline()
+	if !ok {
+		return
+	}
+	now := time.Now()
+	offset := -time.Since(now)
+	until := time.Until(dl)
+	if until > 4500*time.Millisecond {
+		return
+	}
+	fmt.Fprintf(os.Stderr, "[d16] site=%s label=%q now=%v offset=%v timer_clock=%v deadline=%v until=%v ctx_err=%v err=%v\n",
+		site, label, now.Sub(d16Epoch), offset, now.Sub(d16Epoch)-offset, dl.Sub(d16Epoch), until, ctx.Err(), err)
+}
